<script>
const params = new URLSearchParams(location.search);
const titleParam = params.get("title");
const iconParam = params.get("icon");

const emojis = ["📁","📂","🗂️","📌","⭐","🔥","💡","🚧","🧠","⚡","📊","📝","🔖","🎯","🚀","📎"];

function setFavicon(emoji) {
  const svg = `
    <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100">
      <text y="0.9em" font-size="90">${emoji}</text>
    </svg>
  `;
  const url = "data:image/svg+xml," + encodeURIComponent(svg);

  let link = document.querySelector("link[rel~='icon']");
  if (!link) {
    link = document.createElement("link");
    link.rel = "icon";
    document.head.appendChild(link);
  }
  link.href = url;
}

const app = document.getElementById("app");

if (titleParam) {
  const emoji = iconParam || "📁";

  document.title = titleParam;
  setFavicon(emoji);

  app.innerHTML = `
    <div class="preview">
      <span>${emoji}</span>
      ${titleParam}
    </div>
  `;
} else {
  let selectedEmoji = "📁";

  app.innerHTML = `
    <h1>Create Divider</h1>
    <input id="titleInput" placeholder="Enter tab name..." autofocus />

    <div class="emoji-grid" id="emojiGrid"></div>

    <button id="createBtn">Create</button>

    <div class="preview" id="preview">
      <span>📁</span>
      Preview
    </div>
  `;

  const grid = document.getElementById("emojiGrid");
  const preview = document.getElementById("preview");
  const input = document.getElementById("titleInput");

  emojis.forEach(e => {
    const div = document.createElement("div");
    div.className = "emoji";
    div.textContent = e;

    if (e === selectedEmoji) div.classList.add("selected");

    div.onclick = () => {
      selectedEmoji = e;
      document.querySelectorAll(".emoji").forEach(el => el.classList.remove("selected"));
      div.classList.add("selected");
      updatePreview();
    };

    grid.appendChild(div);
  });

  function updatePreview() {
    preview.innerHTML = `
      <span>${selectedEmoji}</span>
      ${input.value || "Preview"}
    `;
  }

  input.addEventListener("input", updatePreview);

  document.getElementById("createBtn").onclick = () => {
    const title = input.value.trim();
    if (!title) return;

    const url = `?title=${encodeURIComponent(title)}&icon=${encodeURIComponent(selectedEmoji)}`;
    location.search = url;
  };
}
</script>
